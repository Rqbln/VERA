  \ifcsname icse@decunits@#2\endcsname
    \expandafter\g@addto@macro\csname icse@decunits@#2\endcsname{, \icse@unitref{#1}}%
  \else
    \expandafter\gdef\csname icse@decunits@#2\endcsname{\icse@unitref{#1}}%
    \global\advance\icse@decunique by 1
  \fi}

  \ifcsname icse@decdone@#2\endcsname\else

    #5\enspace{\footnotesize\color{black!60}--- \icse@ownerlist{#3}, by #4; raised in \csname icse@decunits@#2\endcsname}\par
  \fi}
