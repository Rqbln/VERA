% --- Position arithmetic ---------------------------------------------------
% Position of a saved point in column units: two columns per page, plus the
% fraction of the column height above the point. Differences give pages.
\newcommand{\icse@abspage}[1]{\zref@extractdefault{icse:#1}{abspage}{0}}
\newcommand{\icse@pos}[1]{%
  (2*(\icse@abspage{#1}-1)
   + ((\zposx{icse:#1}/65536 > \paperwidth/2) ? 1 : 0)
   + ((\paperheight - 1in - \voffset - \topmargin - \headheight - \headsep
       - \zposy{icse:#1}/65536) / \textheight))}
\newcommand{\icse@known}[1]{\ifnum\icse@abspage{#1}>0 1\else 0\fi}
\newcommand{\icse@measure}[1]{% pages occupied by unit #1
  \ifcsname icse@next@#1\endcsname
    \edef\icse@nx{\csname icse@next@#1\endcsname}%
    \ifnum\numexpr\icse@known{#1}*\icse@known{\icse@nx}\relax=1
      \fpeval{round((\icse@pos{\icse@nx}-\icse@pos{#1})/2,2)}%
    \else ?\fi
  \else ?\fi}
\newcommand{\icse@delta}[2]{% unit, budget
  \ifcsname icse@next@#1\endcsname
    \edef\icse@nx{\csname icse@next@#1\endcsname}%
    \ifnum\numexpr\icse@known{#1}*\icse@known{\icse@nx}\relax=1
      \edef\icse@d{\fpeval{round((\icse@pos{\icse@nx}-\icse@pos{#1})/2-#2,2)}}%
      \ifnum\fpeval{\icse@d>0.05}=1 \textcolor{pendingred}{\bfseries +\icse@d}%
      \else\ifnum\fpeval{\icse@d<-0.05}=1 \textcolor{brandgreen}{\icse@d}\else 0\fi\fi
    \else\fi
  \fi}
